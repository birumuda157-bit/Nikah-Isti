function uid(){ return Date.now().toString(36)+Math.random().toString(36).slice(2,7); }
function fmtRupiah(n){ n = Number(n||0); return 'Rp' + Math.round(n).toLocaleString('id-ID'); }
function parseNum(v){ const n = Number(String(v||'').replace(/[^\d.-]/g,'')); return isNaN(n)?0:n; }
function fmtDateID(str){ if(!str) return '—'; const d = new Date(str+(str.length<=10?'T00:00:00':'')); if(isNaN(d)) return str; return d.toLocaleDateString('id-ID',{day:'numeric',month:'long',year:'numeric'}); }
function daysUntil(dateStr){ if(!dateStr) return null; const target = new Date(dateStr+'T00:00:00'); const now = new Date(); now.setHours(0,0,0,0); return Math.round((target-now)/86400000); }
function addMonths(dateStr, months){ const d = new Date(dateStr+'T00:00:00'); d.setMonth(d.getMonth()-months); return d; }
function addDaysBefore(dateStr, days){ const d = new Date(dateStr+'T00:00:00'); d.setDate(d.getDate()-days); return d; }
function milestoneDate(m){ if(!S.settings.weddingDate) return null; return m.monthsBefore!=null ? addMonths(S.settings.weddingDate, m.monthsBefore) : addDaysBefore(S.settings.weddingDate, m.daysBefore); }
function esc(s){ return String(s==null?'':s).replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
function logActivity(text){ S.activity.log.unshift({id:uid(), ts:new Date().toISOString(), actor:currentUser==='Pria'?S.settings.groomName:S.settings.brideName, text}); if(S.activity.log.length>150) S.activity.log.length=150; saveKey('activity'); }

/* ============================= COMPUTATIONS ============================= */
function calcSavings(){
  const groom = S.savings.transactions.filter(t=>t.person==='Pria').reduce((a,b)=>a+parseNum(b.amount),0);
  const bride = S.savings.transactions.filter(t=>t.person==='Wanita').reduce((a,b)=>a+parseNum(b.amount),0);
  return {groom, bride, total: groom+bride};
}
function calcBudget(){
  const items = S.budget.items;
  const estimated = items.reduce((a,b)=>a+parseNum(b.estimated),0);
  const actual = items.reduce((a,b)=>a+parseNum(b.actual),0);
  const paid = items.reduce((a,b)=>a+parseNum(b.paid),0);
  const remaining = items.reduce((a,b)=>a+(parseNum(b.actual)-parseNum(b.paid)),0);
  return {estimated, actual, paid, remaining, variance: estimated-actual};
}
function budgetStatus(item){
  const actual = parseNum(item.actual), paid = parseNum(item.paid);
  if(paid<=0) return 'UNPAID'; if(paid>=actual && actual>0) return 'PAID'; return 'PARTIAL';
}
function calcChecklist(){ const items=S.checklist.items; const done=items.filter(i=>i.status==='COMPLETED').length; return {done, total:items.length, pct: items.length? Math.round(done/items.length*100):0}; }
function calcDocuments(){ const items=S.documents.list; const done=items.filter(i=>i.status==='SELESAI').length; return {done, total:items.length, pct: items.length? Math.round(done/items.length*100):0}; }
function calcVendors(){ const items=S.vendors.list; const done=items.filter(i=>['DIPILIH','KONTRAK'].includes(i.status)).length; return {done,total:items.length, pct: items.length? Math.round(done/items.length*100):0}; }
function calcGuests(){
  const list = S.guests.list;
  const invited = list.reduce((a,b)=>a+parseNum(b.invited||1),0);
  const attending = list.filter(g=>g.rsvp==='HADIR').reduce((a,b)=>a+parseNum(b.attending||b.invited||1),0);
  const declined = list.filter(g=>g.rsvp==='TIDAK_HADIR').length;
  const pending = list.filter(g=>g.rsvp==='PENDING').length;
  return {invited, attending, declined, pending, count:list.length};
}
function calcGifts(){
  const list = S.gifts.list;
  const cash = list.filter(g=>g.type==='TUNAI').reduce((a,b)=>a+parseNum(b.amount),0);
  const transfer = list.filter(g=>g.type==='TRANSFER').reduce((a,b)=>a+parseNum(b.amount),0);
  const physical = list.filter(g=>g.type==='BARANG').reduce((a,b)=>a+parseNum(b.amount),0);
  return {cash, transfer, physical, total: cash+transfer+physical};
}
function currentPhase(){
  if(!S.settings.weddingDate) return null;
  const today = new Date(); today.setHours(0,0,0,0);
  let phase = MILESTONES[MILESTONES.length-1];
  for(const m of MILESTONES){ const d = milestoneDate(m); if(d && d>=today){ phase = m; break; } }
  return phase;
}
function preparationScore(){
  const c = calcChecklist(), d = calcDocuments(), v = calcVendors(), g = calcGuests(), b = calcBudget();
  const finPct = b.estimated>0 ? Math.min(100, Math.round(b.paid/b.estimated*100)) : 0;
  const guestPct = g.count>0 ? Math.round(((g.attending+ (g.count-g.pending-g.declined? 0:0))/g.invited||0)*100) : 0;
  const rsvpPct = g.invited>0 ? Math.round((g.attending/g.invited)*100) : 0;
  const parts = [c.pct, d.pct, v.pct, rsvpPct, finPct];
  return Math.round(parts.reduce((a,b)=>a+b,0)/parts.length);
}

function openForm(title, fields, initial, onSubmit){
  const values = Object.assign({}, initial);
  const html = `
    <div class="modal-head"><h3>${esc(title)}</h3><button class="btn ghost icon-only" onclick="App.closeModal()">✕</button></div>
    <form id="dyn-form">
      ${fields.map(f=>{
        if(f.row){
          return `<div class="field-row">${f.row.map(ff=>fieldHtml(ff, values)).join('')}</div>`;
        }
        return fieldHtml(f, values);
      }).join('')}
      <div class="modal-actions">
        <button type="button" class="btn" onclick="App.closeModal()">Batal</button>
        <button type="submit" class="btn primary">Simpan</button>
      </div>
    </form>`;
  document.getElementById('modal-panel').innerHTML = html;
  document.getElementById('modal').classList.add('open');
  document.getElementById('backdrop').classList.add('open');
  document.getElementById('dyn-form').onsubmit = (e)=>{
    e.preventDefault();
    const out = {};
    fields.flatMap(f=>f.row?f.row:[f]).forEach(f=>{
      const el = document.getElementById('f_'+f.key);
      out[f.key] = f.type==='number' ? parseNum(el.value) : el.value;
    });
    onSubmit(out);
    App.closeModal();
  };
}
function fieldHtml(f, values){
  const val = values[f.key]!=null?values[f.key]:'';
  if(f.type==='select'){
    return `<div class="field"><label>${esc(f.label)}</label><select id="f_${f.key}">${f.options.map(o=>`<option value="${esc(o.value)}" ${o.value===val?'selected':''}>${esc(o.label)}</option>`).join('')}</select></div>`;
  }
  if(f.type==='textarea'){
    return `<div class="field"><label>${esc(f.label)}</label><textarea id="f_${f.key}">${esc(val)}</textarea></div>`;
  }
  return `<div class="field"><label>${esc(f.label)}</label><input type="${f.type||'text'}" id="f_${f.key}" value="${esc(val)}" ${f.step?`step="${f.step}"`:''}></div>`;
}
